# Woolworths NZ browse paths, read through the public browse API
# (/api/v1/products?target=browse). robots.txt allows it; only /shop/search is disallowed.
#
# One path per line:  <department>/<aisle>[/<shelf>] [pages=N] [category=label] [covers=a,b,c]
# pages = max pages of 120 products (default 5). covers = extra words that make a
# not-found report fair for search terms belonging here. Lines starting with # are comments.
#
# Watch categories from the app (same set as the Foodstuffs files):
pantry/baking category=baking covers=sugar,flour,baking
pantry/oil-vinegar-condiments category=oil-vinegar-condiments covers=oil,vinegar,sauce,condiment
pantry/tinned-foods-packets category=tinned-foods-packets covers=tinned,canned,tomato,bean
pantry/bulk-foods category=bulk-foods covers=bulk,nut,seed
household/kitchen/kitchen-cleaners category=kitchen-cleaners covers=cleaner,spray
household/bathroom/toilet-bathroom-cleaners category=bathroom-toilet-cleaners covers=toilet,bleach
#
# Stock shelf: liquid stock, stock cubes, pots and powder. The app has no watch category
# for stock, so it is listed here. Without it only matched search results come through,
# and the matcher used to reject names like "Chicken Stock Liquid".
pantry/herbs-spices-stock/stock category=stock covers=stock,broth,bouillon,cube,powder

# --- Fresh herbs, seafood (in-shell mussels) and cooking wine ---
# Wine: the whole cask wine shelf is kept; on white-wine and red-wine only the cheapest 5 per shelf
# by price per litre are kept (scraper/cooking_wine.py).
fruit-veg/fresh-salad-herbs category=fresh-herbs covers=herb,herbs,coriander,cilantro,dill,parsley,basil,mint,chive,chives,thyme,rosemary,sage,salad
fish-seafood/prawns-seafood category=seafood covers=seafood,fish,prawn,shrimp,mussel,shellfish,oyster,crab,squid
fish-seafood/fish category=fish covers=fish,salmon,snapper,hoki,cod,tuna
beer-wine/cask-wine category=cask-wine
beer-wine/white-wine category=white-wine
beer-wine/red-wine category=red-wine

# --- Main grocery shelves, read in full (like Pak'nSave and New World) ---
# `@aisles <department> <aisle,aisle,...|*> [pages=N]` reads each named second-level shelf (aisle) whole,
# with every shelf below it. The paths are looked up in the category tree the run posts to the app, so a
# renamed or removed aisle is logged and skipped, not an error. Promotional aisles (3-for-20, fresh-deals,
# in-season, ...) are never read. pages= is a soft limit: a full last page is read on (up to 20 pages).
# Lines above that name a shelf below one of these aisles are folded into the aisle; their covers= stay.
# `python -m scraper.woolworths_shelves` lists the shelves, expected pages and items without any request.
#
# Beer & wine is left to the cask / cheapest white and red lines above.
@aisles fruit-veg fruit,vegetables,fresh-salad-herbs,prepared-fruit-veg
@aisles meat-poultry *
@aisles fish-seafood *
@aisles fridge-deli cheese,cream-custard,eggs-butter-spreads,milk,yoghurt-desserts,deli-meats-seafood,dips-hummus-nibbles,pasta-pizza-pastry
@aisles bakery *
@aisles pantry *
@aisles frozen frozen-vegetables,frozen-meat,frozen-seafood,frozen-fruit-drink,pizza-pastry-bread
# Household and health-body: the shelves the watch categories point at (cleaning, laundry, kitchen,
# bathroom incl. toilet paper, dental, hair care, bath/shower/soap).
@aisles household cleaning,laundry,kitchen,bathroom
@aisles health-body dental-oral-care,hair-care,bath-shower-soap
